<?php

namespace App\Models;

use CodeIgniter\Model;

class UserModel extends Model
{
    protected $table            = 'users';
    protected $primaryKey       = 'id';
    protected $allowedFields    = [
        'name', 'email', 'password', 'role', 'is_active'
    ];
    protected $returnType       = 'array';
    protected $useTimestamps    = true;

    // Hashear password automáticamente antes de insertar
    protected $beforeInsert     = ['hashPassword'];
    protected $beforeUpdate     = ['hashPassword'];

    protected function hashPassword(array $data): array
    {
        if (isset($data['data']['password'])) {
            $data['data']['password'] = password_hash(
                $data['data']['password'],
                PASSWORD_BCRYPT
            );
        }
        return $data;
    }
}

<?php

namespace App\Filters;

use CodeIgniter\HTTP\RequestInterface;
use CodeIgniter\HTTP\ResponseInterface;
use CodeIgniter\Filters\FilterInterface;

class Cors implements FilterInterface
{
    public function before(RequestInterface $request, $arguments = null)
    {
        // Origen permitido desde .env
        $allowedOrigin = getenv('ALLOWED_ORIGIN');
        $origin        = $request->getHeaderLine('Origin');

        // Validar origen server-side (no confiar solo en el navegador)
        if ($origin !== '' && $origin !== $allowedOrigin) {
            return service('response')
                ->setJSON(['error' => 'Origen no permitido'])
                ->setStatusCode(403);
        }

        // Responder a preflight OPTIONS
        if ($request->getMethod() === 'OPTIONS') {
            return service('response')
                ->setHeader('Access-Control-Allow-Origin', $allowedOrigin)
                ->setHeader('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS')
                ->setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization')
                ->setHeader('Access-Control-Allow-Credentials', 'true')
                ->setStatusCode(204);
        }
    }

    public function after(RequestInterface $request, ResponseInterface $response, $arguments = null)
    {
        $allowedOrigin = getenv('ALLOWED_ORIGIN');

        return $response
            ->setHeader('Access-Control-Allow-Origin', $allowedOrigin)
            ->setHeader('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS')
            ->setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization')
            ->setHeader('Access-Control-Allow-Credentials', 'true');
    }
}
💡
Access-Control-Allow-Credentials: true es obligatorio para que el navegador envíe cookies cross-origin. Con esta opción NO puedes usar * como origen — debe ser el dominio exacto.

<?php

namespace App\Filters;

use CodeIgniter\HTTP\RequestInterface;
use CodeIgniter\HTTP\ResponseInterface;
use CodeIgniter\Filters\FilterInterface;
use Firebase\JWT\JWT;
use Firebase\JWT\Key;

class AuthJWT implements FilterInterface
{
    public function before(RequestInterface $request, $arguments = null)
    {
        // ① Extraer JWT de la cookie HttpOnly (NO de Authorization header)
        $token = $request->getCookie('auth_token');

        if (!$token) {
            return service('response')
                ->setJSON(['error' => 'No autenticado'])
                ->setStatusCode(401);
        }

        try {
            // ② Decodificar y verificar firma + expiración
            $key     = getenv('JWT_SECRET');
            $decoded = JWT::decode($token, new Key($key, 'HS256'));

            // ③ Verificar que el usuario sigue activo en la DB
            $userModel = new \App\Models\UserModel();
            $user = $userModel->where('id', $decoded->sub)
                                  ->where('is_active', 1)
                                  ->first();

            if (!$user) {
                return service('response')
                    ->setJSON(['error' => 'Usuario desactivado'])
                    ->setStatusCode(401);
            }

            // ④ Adjuntar datos del usuario al request
            $request->userData = $decoded;

        } catch (\Firebase\JWT\ExpiredException $e) {
            return service('response')
                ->setJSON(['error' => 'Token expirado'])
                ->setStatusCode(401);
        } catch (\Exception $e) {
            return service('response')
                ->setJSON(['error' => 'Token inválido'])
                ->setStatusCode(401);
        }
    }

    public function after(RequestInterface $request, ResponseInterface $response, $arguments = null)
    {
        // No se necesita acción post-request
    }
}

<?php

namespace App\Controllers;

use CodeIgniter\RESTful\ResourceController;
use App\Models\UserModel;
use Firebase\JWT\JWT;

class Auth extends ResourceController
{
    /**
     * POST /auth/login
     * Recibe email + password, devuelve cookie HttpOnly con JWT
     */
    public function login()
    {
        // ① Validar input
        $rules = [
            'email'    => 'required|valid_email',
            'password' => 'required|min_length[6]',
        ];

        if (!$this->validate($rules)) {
            return $this->respond(['errors' => $this->validator->getErrors()], 422);
        }

        $email    = $this->request->getJSON()->email;
        $password = $this->request->getJSON()->password;

        // ② Buscar usuario activo
        $userModel = new UserModel();
        $user = $userModel->where('email', $email)
                            ->where('is_active', 1)
                            ->first();

        if (!$user || !password_verify($password, $user['password'])) {
            return $this->respond(['error' => 'Credenciales inválidas'], 401);
        }

        // ③ Generar JWT
        $key     = getenv('JWT_SECRET');
        $payload = [
            'sub'  => $user['id'],       // ID del usuario
            'name' => $user['name'],     // Nombre para el frontend
            'role' => $user['role'],     // admin | editor | viewer
            'iat'  => time(),             // Emitido ahora
            'exp'  => time() + 3600,    // Expira en 1 hora
        ];

        $jwt = JWT::encode($payload, $key, 'HS256');

        // ④ Enviar como cookie HttpOnly — CLAVE DE SEGURIDAD
        $this->response->setCookie([
            'name'     => 'auth_token',
            'value'    => $jwt,
            'expire'   => 3600,          // 1 hora
            'httponly'  => true,          // ⛔ JS no puede leerla
            'secure'   => true,          // 🔒 Solo HTTPS
            'samesite' => 'None',        // 🌐 Cross-site (distinto dominio)
            'path'     => '/',
        ]);

        // ⑤ Responder con datos del usuario (sin el JWT)
        return $this->respond([
            'message' => 'Login exitoso',
            'user'    => [
                'id'   => $user['id'],
                'name' => $user['name'],
                'role' => $user['role'],
            ],
        ]);
    }

    /**
     * POST /auth/logout
     * Elimina la cookie de autenticación
     */
    public function logout()
    {
        $this->response->setCookie([
            'name'     => 'auth_token',
            'value'    => '',
            'expire'   => 0,            // Expirar inmediatamente
            'httponly'  => true,
            'secure'   => true,
            'samesite' => 'None',
            'path'     => '/',
        ]);

        return $this->respond(['message' => 'Sesión cerrada']);
    }

    /**
     * GET /auth/me
     * Devuelve datos del usuario actual (requiere authJwt filter)
     */
    public function me()
    {
        $userData = $this->request->userData;

        return $this->respond([
            'user' => [
                'id'   => $userData->sub,
                'name' => $userData->name,
                'role' => $userData->role,
            ],
        ]);
    }
}
⚠️
Se usa SameSite=None porque el frontend (github.io) y el backend (tu-servidor.com) están en dominios distintos. Esto requiere Secure=true (HTTPS obligatorio en ambos). Si estuvieran en el mismo dominio, usarías SameSite=Strict para mayor seguridad.

<?php

namespace App\Controllers;

use CodeIgniter\RESTful\ResourceController;

class ApiProxy extends ResourceController
{
    // Credenciales que SOLO existen en el servidor
    private string $apiUrl;
    private string $apiKey;

    public function __construct()
    {
        $this->apiUrl = getenv('API_EXTERNAL_URL');
        $this->apiKey = getenv('API_EXTERNAL_KEY');
    }

    /**
     * Helper — Headers para la API externa
     */
    private function apiHeaders(): array
    {
        return [
            'headers' => [
                'Authorization' => 'Bearer ' . $this->apiKey,
                'Content-Type'  => 'application/json',
                'Accept'        => 'application/json',
            ],
        ];
    }

    /**
     * Helper — Verificar rol mínimo requerido
     */
    private function requireRole(string $minimumRole): bool
    {
        $roles    = ['viewer' => 1, 'editor' => 2, 'admin' => 3];
        $userRole = $this->request->userData->role;

        return ($roles[$userRole] ?? 0) >= ($roles[$minimumRole] ?? 99);
    }

    // ─────────────────────────────────────────────────
    // GET /proxy/items — Listar (viewer+)
    // ─────────────────────────────────────────────────
    public function index()
    {
        $client   = \Config\Services::curlrequest();
        $response = $client->get(
            $this->apiUrl . '/items',
            $this->apiHeaders()
        );

        return $this->respond(
            json_decode($response->getBody()),
            $response->getStatusCode()
        );
    }

    // ─────────────────────────────────────────────────
    // GET /proxy/items/:id — Detalle (viewer+)
    // ─────────────────────────────────────────────────
    public function show($id = null)
    {
        $client   = \Config\Services::curlrequest();
        $response = $client->get(
            $this->apiUrl . "/items/{$id}",
            $this->apiHeaders()
        );

        return $this->respond(
            json_decode($response->getBody()),
            $response->getStatusCode()
        );
    }

    // ─────────────────────────────────────────────────
    // POST /proxy/items — Crear (editor+)
    // ─────────────────────────────────────────────────
    public function create()
    {
        if (!$this->requireRole('editor')) {
            return $this->respond(['error' => 'Sin permisos para crear'], 403);
        }

        $body     = $this->request->getJSON(true);
        $client   = \Config\Services::curlrequest();
        $options  = array_merge($this->apiHeaders(), ['json' => $body]);
        $response = $client->post($this->apiUrl . '/items', $options);

        return $this->respond(
            json_decode($response->getBody()),
            $response->getStatusCode()
        );
    }

    // ─────────────────────────────────────────────────
    // PUT /proxy/items/:id — Actualizar (editor+)
    // ─────────────────────────────────────────────────
    public function update($id = null)
    {
        if (!$this->requireRole('editor')) {
            return $this->respond(['error' => 'Sin permisos para editar'], 403);
        }

        $body     = $this->request->getJSON(true);
        $client   = \Config\Services::curlrequest();
        $options  = array_merge($this->apiHeaders(), ['json' => $body]);
        $response = $client->put($this->apiUrl . "/items/{$id}", $options);

        return $this->respond(
            json_decode($response->getBody()),
            $response->getStatusCode()
        );
    }

    // ─────────────────────────────────────────────────
    // DELETE /proxy/items/:id — Eliminar (admin ONLY)
    // ─────────────────────────────────────────────────
    public function delete($id = null)
    {
        if (!$this->requireRole('admin')) {
            return $this->respond(['error' => 'Solo administradores pueden eliminar'], 403);
        }

        $client   = \Config\Services::curlrequest();
        $response = $client->delete(
            $this->apiUrl . "/items/{$id}",
            $this->apiHeaders()
        );

        return $this->respond(
            json_decode($response->getBody()),
            $response->getStatusCode()
        );
    }
}
✅
Las credenciales de la API externa (API_EXTERNAL_KEY) nunca llegan al frontend. Solo el proxy las conoce. El control de roles impide que un viewer ejecute DELETE aunque tenga sesión válida.

// ─── Base URL del proxy (NO de la API externa) ─────
const BASE_URL = import.meta.env.VITE_API_BASE_URL;
// Ejemplo: https://tu-servidor.com

// ─── Tipos ──────────────────────────────────────────
interface User {
  id: number;
  name: string;
  role: 'admin' | 'editor' | 'viewer';
}

interface LoginResponse {
  message: string;
  user: User;
}

// ─── Fetch wrapper con credentials ──────────────────
async function apiFetch<T>(
  endpoint: string,
  options: RequestInit = {}
): Promise<T> {
  const response = await fetch(`${BASE_URL}${endpoint}`, {
    ...options,
    credentials: 'include',  // ← CLAVE: envía la cookie automáticamente
    headers: {
      'Content-Type': 'application/json',
      ...options.headers,
    },
  });

  if (response.status === 401) {
    // Token expirado o inválido → redirigir a login
    window.location.href = '/login';
    throw new Error('No autenticado');
  }

  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.error || 'Error en la petición');
  }

  return response.json();
}

// ─── Auth endpoints ─────────────────────────────────
export const authApi = {
  login: (email: string, password: string) =>
    apiFetch<LoginResponse>('/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password }),
    }),

  logout: () =>
    apiFetch('/auth/logout', { method: 'POST' }),

  me: () =>
    apiFetch<{ user: User }>('/auth/me'),
};

// ─── Proxy API endpoints ────────────────────────────
export const itemsApi = {
  getAll: () =>
    apiFetch<any[]>('/proxy/items'),

  getById: (id: number) =>
    apiFetch<any>(`/proxy/items/${id}`),

  create: (data: Record<string, unknown>) =>
    apiFetch('/proxy/items', {
      method: 'POST',
      body: JSON.stringify(data),
    }),

  update: (id: number, data: Record<string, unknown>) =>
    apiFetch(`/proxy/items/${id}`, {
      method: 'PUT',
      body: JSON.stringify(data),
    }),

  remove: (id: number) =>
    apiFetch(`/proxy/items/${id}`, {
      method: 'DELETE',
    }),
};
💡
Nota que no hay tokens, API keys ni credenciales en ningún lugar del código frontend. Solo credentials: 'include' y el navegador se encarga de adjuntar la cookie.

import { createContext, useContext, useState, useEffect } from 'react';
import { authApi } from '../services/apiClient';

interface User {
  id: number;
  name: string;
  role: 'admin' | 'editor' | 'viewer';
}

interface AuthContextType {
  user: User | null;
  loading: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType>(null!);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  // Al montar, verificar si ya hay sesión activa (cookie válida)
  useEffect(() => {
    authApi.me()
      .then((res) => setUser(res.user))
      .catch(() => setUser(null))
      .finally(() => setLoading(false));
  }, []);

  const login = async (email: string, password: string) => {
    const res = await authApi.login(email, password);
    setUser(res.user);  // Cookie ya fue seteada por el backend
  };

  const logout = async () => {
    await authApi.logout();
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, loading, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);

import { useState } from 'react';
import { useAuth } from '../context/AuthContext';
import { useNavigate } from 'react-router-dom';

export default function LoginPage() {
  const [email, setEmail]       = useState('');
  const [password, setPassword] = useState('');
  const [error, setError]       = useState('');
  const [loading, setLoading]   = useState(false);
  const { login } = useAuth();
  const navigate = useNavigate();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    setLoading(true);

    try {
      await login(email, password);
      navigate('/dashboard');
    } catch (err) {
      setError('Email o contraseña incorrectos');
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <h1>Iniciar Sesión</h1>

      {error && <div className="error">{error}</div>}

      <input
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        placeholder="Email"
        required
      />

      <input
        type="password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
        placeholder="Contraseña"
        required
      />

      <button type="submit" disabled={loading}>
        {loading ? 'Ingresando...' : 'Ingresar'}
      </button>
    </form>
  );
}

import { Navigate } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

interface Props {
  children: React.ReactNode;
  requiredRole?: 'admin' | 'editor' | 'viewer';
}

export default function ProtectedRoute({ children, requiredRole }: Props) {
  const { user, loading } = useAuth();

  if (loading) return <div>Cargando...</div>;
  if (!user)  return <Navigate to="/login" />;

  // Verificar rol si es necesario
  if (requiredRole) {
    const roles = { viewer: 1, editor: 2, admin: 3 };
    if (roles[user.role] < roles[requiredRole]) {
      return <div>No tienes permisos para esta sección</div>;
    }
  }

  return <>{children}</>;
}

import { useState, useEffect } from 'react';
import { useAuth } from '../context/AuthContext';
import { itemsApi } from '../services/apiClient';

export default function Dashboard() {
  const { user, logout } = useAuth();
  const [items, setItems] = useState<any[]>([]);

  useEffect(() => {
    // La cookie se envía automáticamente — sin tokens manuales
    itemsApi.getAll().then(setItems);
  }, []);

  const handleDelete = async (id: number) => {
    try {
      await itemsApi.remove(id);
      setItems(items.filter(i => i.id !== id));
    } catch (err) {
      alert('No tienes permisos para eliminar');
    }
  };

  return (
    <div>
      <h1>Dashboard</h1>
      <p>Hola {user?.name} — Rol: {user?.role}</p>
      <button onClick={logout}>Cerrar sesión</button>

      {items.map(item => (
        <div key={item.id}>
          <span>{item.name}</span>
          {user?.role === 'admin' && (
            <button onClick={() => handleDelete(item.id)}>
              Eliminar
            </button>
          )}
        </div>
      ))}
    </div>

.github/workflows/deploy.yml
YAML 
name: Deploy to GitHub Pages

on:
  push:
    branches: [main]

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'npm'

      - run: npm ci

      # Build con la URL del proxy inyectada desde GitHub Secrets
      - run: npm run build
        env:
          VITE_API_BASE_URL: ${{ secrets.VITE_API_BASE_URL }}
          # Solo la URL del proxy, NO credenciales de la API

      - uses: actions/upload-pages-artifact@v3
        with:
          path: './dist'

      - uses: actions/deploy-pages@v4
        id: deployment
✅
El único secreto en GitHub Actions es VITE_API_BASE_URL (la URL de tu servidor Apache). Las credenciales de la API externa viven exclusivamente en el .env del servidor — nunca en el repositorio ni en el frontend.
